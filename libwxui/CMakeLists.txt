cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(libwxui LANGUAGES CXX)
endif()
find_package(fmt CONFIG REQUIRED)
find_path(WXUI_RAPIDJSON_INCLUDE rapidjson/document.h REQUIRED)
find_path(WXUI_UTFCPP_INCLUDE utf8.h PATH_SUFFIXES utf8cpp utfcpp REQUIRED)
find_package(wxWidgets CONFIG QUIET COMPONENTS core base xml aui html stc)
add_library(wxui_dependencies INTERFACE)
if(TARGET wx::core)
  target_link_libraries(wxui_dependencies INTERFACE wx::core wx::base wx::xml wx::aui wx::html wx::stc)
else()
  find_package(wxWidgets REQUIRED COMPONENTS core base xml aui html stc)
  target_include_directories(wxui_dependencies INTERFACE ${wxWidgets_INCLUDE_DIRS})
  target_compile_definitions(wxui_dependencies INTERFACE ${wxWidgets_DEFINITIONS})
  separate_arguments(WXUI_WX_CXX_FLAGS NATIVE_COMMAND "${wxWidgets_CXX_FLAGS}")
  target_compile_options(wxui_dependencies INTERFACE ${WXUI_WX_CXX_FLAGS})
  target_link_libraries(wxui_dependencies INTERFACE ${wxWidgets_LIBRARIES})
endif()
# Resolve cached Apple SDK libm paths using the active toolchain.
# Some NanoSVG binary packages retain the producer's SDK path.
if(APPLE AND TARGET NanoSVG::nanosvg)
  get_target_property(wxui_svg_links NanoSVG::nanosvg INTERFACE_LINK_LIBRARIES)
  if(wxui_svg_links)
    list(TRANSFORM wxui_svg_links REPLACE ".*/MacOSX[^/]*[.]sdk/usr/lib/libm[.]tbd$" "m")
    set_property(TARGET NanoSVG::nanosvg PROPERTY INTERFACE_LINK_LIBRARIES "${wxui_svg_links}")
  endif()
endif()
# Standalone build of the libwxui snapshot supplied with this repository.
# No OrbitBridge helper macros, directory globals or private libraries.
file(GLOB_RECURSE sources CONFIGURE_DEPENDS src/*.cc)
file(GLOB_RECURSE headers CONFIGURE_DEPENDS include/*.hpp include/*.h src/*.hpp src/*.h)
add_library(libwxui STATIC ${sources} ${headers})
if(APPLE)
  enable_language(OBJCXX)
  target_sources(libwxui PRIVATE src/window_drag.mm)
  target_link_libraries(libwxui PRIVATE "-framework AppKit")
endif()
source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${sources} ${headers})
source_group(build FILES "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
target_include_directories(libwxui PUBLIC include PRIVATE "${WXUI_RAPIDJSON_INCLUDE}" "${WXUI_UTFCPP_INCLUDE}")
target_link_libraries(libwxui PUBLIC wxui_dependencies fmt::fmt)
target_compile_features(libwxui PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(libwxui PRIVATE /utf-8 /FS)
endif()

option(LIBWXUI_BUILD_TESTING "Build focused desktop services tests (requires a display)" OFF)
if(LIBWXUI_BUILD_TESTING)
  enable_testing()
  add_executable(wxui_desktop_services_test tests/desktop_services_test.cc)
  target_link_libraries(wxui_desktop_services_test PRIVATE libwxui)
  add_test(NAME libwxui.desktop_services COMMAND wxui_desktop_services_test)
  set_tests_properties(libwxui.desktop_services PROPERTIES TIMEOUT 15 ENVIRONMENT "TZ=America/New_York")
endif()
