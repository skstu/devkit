cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/NativeLibsBuild.cmake")
project(devkit_ble VERSION "${NATIVE_LIBS_VERSION}" LANGUAGES C CXX)
if(NOT APPLE)
  message(FATAL_ERROR "libble currently implements the Apple backend only; other backends remain pending.")
endif()
enable_language(Swift)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(ble_swift src/apple/BleTypes.swift src/apple/BleEngine.swift src/apple/BleABI.swift)
if(NOT CMAKE_OSX_ARCHITECTURES)
  set(CMAKE_OSX_ARCHITECTURES "${CMAKE_SYSTEM_PROCESSOR}")
endif()
if(NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(ble_target "${CMAKE_OSX_ARCHITECTURES}-apple-ios${CMAKE_OSX_DEPLOYMENT_TARGET}")
  if(CMAKE_OSX_SYSROOT MATCHES "simulator")
    string(APPEND ble_target "-simulator")
  endif()
else()
  set(ble_target "${CMAKE_OSX_ARCHITECTURES}-apple-macosx${CMAKE_OSX_DEPLOYMENT_TARGET}")
endif()
function(ble_swift_settings target)
  target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:Swift>:-target;${ble_target}>")
  target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:Swift>:-swift-version;5;-import-objc-header;${CMAKE_CURRENT_SOURCE_DIR}/include/libble/ble.h;-Xcc;-DDKBLE_TYPES_ONLY>")
  target_link_libraries(${target} PRIVATE "-framework Foundation" "-framework CoreBluetooth")
endfunction()
add_library(devkit_ble SHARED ${ble_swift} src/version.c)
add_library(devkit::ble ALIAS devkit_ble)
ble_swift_settings(devkit_ble)
target_compile_definitions(devkit_ble PRIVATE "$<$<COMPILE_LANGUAGE:C>:DEVKIT_BLE_BUILD=1;DKBLE_VERSION=\"${NATIVE_LIBS_VERSION}\">")
target_include_directories(devkit_ble PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
set_target_properties(devkit_ble PROPERTIES EXPORT_NAME ble OUTPUT_NAME devkit_ble VERSION ${NATIVE_LIBS_VERSION} SOVERSION 1 INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR ON C_VISIBILITY_PRESET hidden)
target_link_options(devkit_ble PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/src/exports.txt")
set_property(TARGET devkit_ble APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/exports.txt")
install(TARGETS devkit_ble EXPORT devkitBleTargets LIBRARY DESTINATION lib RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(FILES include/libble/ble.h DESTINATION include/libble)
install(FILES LICENSE README.md DESTINATION share/libble)
install(DIRECTORY docs/ DESTINATION share/libble/docs)
install(DIRECTORY examples/consumer DESTINATION share/libble/examples)
install(EXPORT devkitBleTargets NAMESPACE devkit:: DESTINATION lib/cmake/devkitBle)
configure_package_config_file(cmake/devkitBleConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/devkitBleConfig.cmake" INSTALL_DESTINATION lib/cmake/devkitBle)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/devkitBleConfigVersion.cmake" VERSION ${NATIVE_LIBS_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/devkitBleConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/devkitBleConfigVersion.cmake" cmake/BundleRuntime.cmake DESTINATION lib/cmake/devkitBle)
if(BUILD_TESTING)
  add_executable(ble_abi_test tests/abi.c)
  target_link_libraries(ble_abi_test PRIVATE devkit_ble)
  add_test(NAME libble.c_abi COMMAND ble_abi_test)
  add_executable(ble_state_test ${ble_swift} tests/state.swift)
  ble_swift_settings(ble_state_test)
  target_compile_options(ble_state_test PRIVATE -DLIBBLE_TESTING)
  add_test(NAME libble.extracted_state COMMAND ble_state_test)
  set_tests_properties(libble.c_abi libble.extracted_state PROPERTIES TIMEOUT 20)
endif()
