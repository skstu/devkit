cmake_minimum_required(VERSION 3.21)
include(cmake/NativeLibsBuild.cmake)
set(NATIVE_LIBS_COMPONENTS "libpath;libstl;libcompr" CACHE STRING "Components to build")
option(LIBPATH_UTF8_ONLY "Use only standard UTF-8 paths" ON)
option(LIBCOMPR_STREAM_ONLY "Use only bounded streaming utilities" ON)
option(LIBSYS_COMPONENTS_ONLY "Build small system components without the platform facade" ON)

# Select vcpkg features before project() loads its toolchain. Only this root
# manifest is affected; embedded consumers own their dependency resolution.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  foreach(component IN LISTS NATIVE_LIBS_COMPONENTS)
    if(component MATCHES "^(libcrypt|libnet|libsys|liblog|libdb|libuvbrg|liburl|libwxui|libfix)$")
      list(APPEND VCPKG_MANIFEST_FEATURES "${component}")
    endif()
  endforeach()
  if(NOT LIBPATH_UTF8_ONLY)
    list(APPEND VCPKG_MANIFEST_FEATURES legacy-path)
  endif()
  list(REMOVE_DUPLICATES VCPKG_MANIFEST_FEATURES)
endif()
project(NativeLibs VERSION "${NATIVE_LIBS_VERSION}" LANGUAGES C CXX)
include(GNUInstallDirs)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  include(CTest)
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

set(native_libs_supported libpath libstl libcompr libsys libcrypt libnet liblog libdb libuvbrg liburl libwxui libfix)
foreach(component IN LISTS NATIVE_LIBS_COMPONENTS)
  if(NOT component IN_LIST native_libs_supported)
    message(FATAL_ERROR "Unsupported component: ${component}; libengjs is retained legacy code")
  endif()
endforeach()
if("libcompr" IN_LIST NATIVE_LIBS_COMPONENTS AND NOT LIBCOMPR_STREAM_ONLY)
  message(FATAL_ERROR "The legacy full libcompr archive backend is not yet supported by this standalone entry")
endif()
# Stable dependency order, regardless of the user's component list ordering.
foreach(component IN LISTS native_libs_supported)
  if(component IN_LIST NATIVE_LIBS_COMPONENTS)
    add_subdirectory("${component}")
  endif()
endforeach()
if(BUILD_TESTING AND TARGET libcrypt AND TARGET libstl_core AND TARGET libcompr_stream)
  add_executable(native_libs_foundation_test tests/foundation_reuse_test.cpp)
  target_compile_features(native_libs_foundation_test PRIVATE cxx_std_20)
  target_include_directories(native_libs_foundation_test PRIVATE libnet/include)
  target_link_libraries(native_libs_foundation_test PRIVATE libcrypt libstl_core libcompr_stream)
  add_test(NAME native-libs.foundation COMMAND native_libs_foundation_test)
endif()
