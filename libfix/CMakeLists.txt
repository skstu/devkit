cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(libfix LANGUAGES C CXX)
  include(CTest)
endif()
include(GNUInstallDirs)
find_package(quickfix CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
if(NOT TARGET libnet_uv)
  include("${CMAKE_CURRENT_LIST_DIR}/../libnet/cmake/Uv.cmake")
endif()
add_library(libfix STATIC src/endpoint.cc)
add_library(devkit::libfix ALIAS libfix)
target_compile_features(libfix PUBLIC cxx_std_20)
set_target_properties(libfix PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_include_directories(libfix PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(libfix PUBLIC quickfix PRIVATE libnet_uv OpenSSL::SSL Threads::Threads)
if(MSVC)
  target_compile_options(libfix PRIVATE /W4 /permissive-)
  target_compile_definitions(libfix PRIVATE NOMINMAX)
else()
  target_compile_options(libfix PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS libfix ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(DIRECTORY include/libfix DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(FILES LICENSE NOTICE.md DESTINATION "${CMAKE_INSTALL_DATADIR}/devkit/libfix")
if(BUILD_TESTING)
  add_executable(libfix_transport_test tests/transport.cpp)
  target_link_libraries(libfix_transport_test PRIVATE libfix OpenSSL::SSL)
  add_test(NAME devkit.libfix.transport COMMAND libfix_transport_test)
  set_tests_properties(devkit.libfix.transport PROPERTIES TIMEOUT 30)
endif()
