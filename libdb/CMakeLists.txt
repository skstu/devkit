cmake_minimum_required(VERSION 3.21)

find_package(sqlcipher 4.19.0 EXACT CONFIG REQUIRED)
if(NOT SQLCIPHER_SOVKIT_BUILD_PROFILE STREQUAL "4.19.0-openssl-static-v1")
  message(FATAL_ERROR "libdb requires SovranKit's pinned static SQLCipher overlay")
endif()
add_library(libdb STATIC src/libdb.cpp)
add_library(sovrankit::libdb ALIAS libdb)
set_target_properties(libdb PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
target_compile_features(libdb PUBLIC cxx_std_20)
target_include_directories(libdb PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_link_libraries(libdb PRIVATE sqlcipher::sqlcipher)
if(COMMAND sovrankit_set_warnings)
  sovrankit_set_warnings(libdb)
endif()
install(TARGETS libdb ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(DIRECTORY include/ DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
