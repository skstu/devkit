cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/NativeLibsBuild.cmake")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(devkit_ice VERSION "${NATIVE_LIBS_VERSION}" LANGUAGES C CXX)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(Threads REQUIRED)
find_package(LibJuice 1.7.2 EXACT CONFIG REQUIRED)
find_package(libuv CONFIG REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
get_target_property(juice_type LibJuice::LibJuice TYPE)
if(NOT juice_type STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "The libice SDK requires the statically linked pinned libjuice overlay.")
endif()
add_library(devkit_ice SHARED src/ice_api.cc ../libnet/src/libnet_ice.cc ../libnet/src/libnet_network_path.cc ../libnet/src/libnet_uv.cc)
add_library(devkit::ice ALIAS devkit_ice)
target_compile_features(devkit_ice PRIVATE cxx_std_20)
target_compile_definitions(devkit_ice PRIVATE DEVKIT_ICE_BUILD=1 DKICE_VERSION="${NATIVE_LIBS_VERSION}")
target_include_directories(devkit_ice PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE ../libnet/include)
target_link_libraries(devkit_ice PRIVATE LibJuice::LibJuice libuv::uv_a OpenSSL::Crypto)
set_target_properties(devkit_ice PROPERTIES EXPORT_NAME ice OUTPUT_NAME devkit_ice VERSION ${NATIVE_LIBS_VERSION} SOVERSION 1 CXX_VISIBILITY_PRESET hidden C_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
native_libs_set_warnings(devkit_ice)
if(APPLE)
  set_target_properties(devkit_ice PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR ON)
  target_link_options(devkit_ice PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/src/exports.txt" "LINKER:-dead_strip")
  set_property(TARGET devkit_ice APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/exports.txt")
elseif(UNIX)
  target_link_options(devkit_ice PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map" "LINKER:--exclude-libs,ALL" -static-libstdc++ -static-libgcc)
  set_property(TARGET devkit_ice APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map")
endif()
if(ANDROID)
  target_link_libraries(devkit_ice PRIVATE android)
elseif(WIN32)
  target_link_libraries(devkit_ice PRIVATE iphlpapi)
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/provider-inputs.json" CONTENT "{\"libjuice\":\"$<TARGET_FILE:LibJuice::LibJuice>\",\"libuv\":\"$<TARGET_FILE:libuv::uv_a>\",\"openssl\":\"$<TARGET_FILE:OpenSSL::Crypto>\"}\n")
install(TARGETS devkit_ice EXPORT devkitIceTargets LIBRARY DESTINATION lib RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(FILES include/libice/ice.h DESTINATION include/libice)
install(FILES LICENSE README.md DESTINATION share/libice)
install(DIRECTORY docs/ DESTINATION share/libice/docs)
install(DIRECTORY examples/consumer DESTINATION share/libice/examples)
install(EXPORT devkitIceTargets NAMESPACE devkit:: DESTINATION lib/cmake/devkitIce)
configure_package_config_file(cmake/devkitIceConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/devkitIceConfig.cmake" INSTALL_DESTINATION lib/cmake/devkitIce)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/devkitIceConfigVersion.cmake" VERSION ${NATIVE_LIBS_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/devkitIceConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/devkitIceConfigVersion.cmake" cmake/BundleRuntime.cmake DESTINATION lib/cmake/devkitIce)
if(BUILD_TESTING)
  add_executable(ice_abi_test tests/abi.c)
  target_link_libraries(ice_abi_test PRIVATE devkit_ice)
  add_test(NAME libice.c_abi COMMAND ice_abi_test)
  add_executable(ice_local_test tests/local.cpp)
  target_compile_features(ice_local_test PRIVATE cxx_std_20)
  target_link_libraries(ice_local_test PRIVATE devkit_ice LibJuice::LibJuice)
  add_test(NAME libice.local_transport COMMAND ice_local_test)
  set_tests_properties(libice.c_abi libice.local_transport PROPERTIES TIMEOUT 30)
endif()
