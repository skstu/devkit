cmake_minimum_required(VERSION 3.21)
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/../../VERSION" DEVKIT_VERSION LIMIT_COUNT 1)
project(devkitDevice VERSION ${DEVKIT_VERSION} LANGUAGES CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(device_info STATIC ../src/device_info.cc)
add_library(devkit::device_info ALIAS device_info)
set_target_properties(device_info PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
target_compile_features(device_info PUBLIC cxx_std_20)
target_include_directories(device_info PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/../include> $<INSTALL_INTERFACE:include>)
if(MSVC)
  set_property(TARGET device_info PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(device_info PRIVATE /W4 /utf-8)
else()
  target_compile_options(device_info PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  target_link_libraries(device_info PUBLIC "-framework IOKit" "-framework CoreFoundation")
endif()
install(TARGETS device_info EXPORT devkitDeviceTargets ARCHIVE DESTINATION lib)
install(FILES ../include/libsys/device_info.h DESTINATION include/libsys)
install(EXPORT devkitDeviceTargets NAMESPACE devkit:: DESTINATION lib/cmake/devkitDevice)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/devkitDeviceConfig.cmake" "include(\"\${CMAKE_CURRENT_LIST_DIR}/devkitDeviceTargets.cmake\")\n")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/devkitDeviceConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/devkitDeviceConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/devkitDeviceConfigVersion.cmake" DESTINATION lib/cmake/devkitDevice)
include(CTest)
if(BUILD_TESTING)
  add_executable(device_info_test ../tests/device_info_test.cpp)
  target_link_libraries(device_info_test PRIVATE device_info)
  add_test(NAME libsys.device_info COMMAND device_info_test)
endif()
