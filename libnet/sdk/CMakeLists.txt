cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/NativeLibsBuild.cmake")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(devkit_net VERSION "${NATIVE_LIBS_VERSION}" LANGUAGES C CXX)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(libuv CONFIG REQUIRED)
find_package(ngtcp2 CONFIG REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS SSL Crypto)
add_library(devkit_net SHARED ../src/net_api.cc ../src/lan_api.cc ../src/libnet_uv.cc ../src/libnet_quic.cc)
add_library(devkit::net ALIAS devkit_net)
target_compile_features(devkit_net PRIVATE cxx_std_20)
target_compile_definitions(devkit_net PRIVATE DEVKIT_NET_BUILD=1 DKNET_VERSION="${NATIVE_LIBS_VERSION}")
target_include_directories(devkit_net PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/../include> $<INSTALL_INTERFACE:include>)
target_link_libraries(devkit_net PRIVATE libuv::uv_a ngtcp2::ngtcp2_static ngtcp2::ngtcp2_crypto_ossl_static OpenSSL::SSL OpenSSL::Crypto)
set_target_properties(devkit_net PROPERTIES EXPORT_NAME net OUTPUT_NAME devkit_net VERSION ${NATIVE_LIBS_VERSION} SOVERSION 1 CXX_VISIBILITY_PRESET hidden C_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(WIN32)
  target_link_libraries(devkit_net PRIVATE iphlpapi)
endif()
native_libs_set_warnings(devkit_net)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set_target_properties(devkit_net PROPERTIES FRAMEWORK TRUE OUTPUT_NAME DevkitNet
    MACOSX_FRAMEWORK_IDENTIFIER com.skstu.devkit.libnet
    MACOSX_FRAMEWORK_BUNDLE_VERSION "${NATIVE_LIBS_VERSION}"
    MACOSX_FRAMEWORK_SHORT_VERSION_STRING "${NATIVE_LIBS_VERSION}"
    XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
endif()
if(APPLE)
  set_target_properties(devkit_net PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR ON)
  target_link_options(devkit_net PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_LIST_DIR}/exports.txt")
  set_property(TARGET devkit_net APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/exports.txt")
elseif(UNIX)
  target_link_options(devkit_net PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_LIST_DIR}/exports.map" "LINKER:--exclude-libs,ALL" -static-libstdc++)
  if(ANDROID)
    target_link_options(devkit_net PRIVATE -static-libgcc)
  else()
    # A host callback may throw into our exception barrier. ELF DSOs must
    # share the unwinder; a hidden static libgcc aborts on foreign frames.
    target_link_options(devkit_net PRIVATE -shared-libgcc)
  endif()
  set_property(TARGET devkit_net APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/exports.map")
endif()
install(TARGETS devkit_net EXPORT devkitNetTargets LIBRARY DESTINATION lib RUNTIME DESTINATION bin ARCHIVE DESTINATION lib FRAMEWORK DESTINATION lib)
install(FILES ../include/libnet/net.h DESTINATION include/libnet)
install(FILES ../LICENSE ../docs/SDK.md ../docs/LAN.md DESTINATION share/libnet)
install(DIRECTORY ../examples/consumer DESTINATION share/libnet/examples)
install(EXPORT devkitNetTargets NAMESPACE devkit:: DESTINATION lib/cmake/devkitNet)
configure_package_config_file(devkitNetConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/devkitNetConfig.cmake" INSTALL_DESTINATION lib/cmake/devkitNet)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/devkitNetConfigVersion.cmake" VERSION ${NATIVE_LIBS_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/devkitNetConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/devkitNetConfigVersion.cmake" BundleRuntime.cmake DESTINATION lib/cmake/devkitNet)
if(BUILD_TESTING)
  add_executable(net_abi_test ../tests/abi.c)
  target_link_libraries(net_abi_test PRIVATE devkit_net)
  add_test(NAME libnet.c_abi COMMAND net_abi_test)
  add_executable(net_lifecycle_test ../tests/lifecycle.cpp)
  target_compile_features(net_lifecycle_test PRIVATE cxx_std_20)
  target_link_libraries(net_lifecycle_test PRIVATE devkit_net)
  add_test(NAME libnet.lifecycle COMMAND net_lifecycle_test)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_test(NAME libnet.receive_burst COMMAND net_lifecycle_test ReceiveBurst)
    set_tests_properties(libnet.receive_burst PROPERTIES TIMEOUT 5)
  endif()
  add_executable(net_lan_test ../tests/lan.c)
  target_link_libraries(net_lan_test PRIVATE devkit_net)
  add_test(NAME libnet.lan COMMAND net_lan_test)
  add_executable(net_lan_lifecycle_test ../tests/lan_lifecycle.cpp)
  target_compile_features(net_lan_lifecycle_test PRIVATE cxx_std_20)
  target_include_directories(net_lan_lifecycle_test PRIVATE ../src)
  target_link_libraries(net_lan_lifecycle_test PRIVATE devkit_net)
  add_test(NAME libnet.lan_lifecycle COMMAND net_lan_lifecycle_test)
  if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    add_executable(net_lan_send_apple_test ../tests/lan_send_apple.cpp)
    target_compile_features(net_lan_send_apple_test PRIVATE cxx_std_20)
    target_link_libraries(net_lan_send_apple_test PRIVATE devkit_net)
    add_test(NAME libnet.lan_send_apple COMMAND net_lan_send_apple_test)
    set_tests_properties(libnet.lan_send_apple PROPERTIES TIMEOUT 5)
  endif()
  set_tests_properties(libnet.c_abi libnet.lifecycle libnet.lan libnet.lan_lifecycle PROPERTIES TIMEOUT 30)
endif()
