cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(flui_quotes_consumer LANGUAGES CXX)
  find_package(libflui CONFIG REQUIRED)
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS flui.xml wxui.xml)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/flui.xml" FLUI_XML)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/wxui.xml" WXUI_XML)
configure_file(layouts.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/layouts.hpp" @ONLY)
add_executable(flui_quotes MACOSX_BUNDLE flui_main.cpp)
target_compile_features(flui_quotes PRIVATE cxx_std_17)
target_include_directories(flui_quotes PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(flui_quotes PRIVATE devkit::libflui)
set_target_properties(flui_quotes PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.skstu.libflui.quotes MACOSX_BUNDLE_BUNDLE_NAME "libflui Quotes")
libflui_bundle_runtime(flui_quotes)
if(LIBFLUI_BUILD_WXUI_SAMPLE)
  if(NOT TARGET libwxui)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../libwxui" "${CMAKE_CURRENT_BINARY_DIR}/wxui")
  endif()
  add_executable(wxui_quotes MACOSX_BUNDLE wxui_main.cpp)
  target_include_directories(wxui_quotes PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(wxui_quotes PRIVATE libwxui)
  set_target_properties(wxui_quotes PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.skstu.libflui.wxquotes MACOSX_BUNDLE_BUNDLE_NAME "libwxui Quotes")
  if(BUILD_TESTING)
    add_test(NAME libflui.wxui_controller COMMAND wxui_quotes --self-test)
    set_tests_properties(libflui.wxui_controller PROPERTIES TIMEOUT 20)
  endif()
endif()
