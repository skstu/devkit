cmake_minimum_required(VERSION 3.24)
# CMake reads the deployment target while enabling the compiler. A target
# property named OSX_DEPLOYMENT_TARGET does not set the compiler minimum.
if(CMAKE_HOST_APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS" AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS version")
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(libflui LANGUAGES C CXX)
  include(CTest)
endif()
option(LIBFLUI_PORTABLE_HOST "Build the SDK-owned single-surface host bridge" OFF)
if(LIBFLUI_PORTABLE_HOST)
  include(GNUInstallDirs)
  include(CMakePackageConfigHelpers)
  file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" LIBFLUI_VERSION LIMIT_COUNT 1)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_library(libflui SHARED src/host_bridge.cpp)
  add_library(devkit::libflui ALIAS libflui)
  target_compile_features(libflui PRIVATE cxx_std_17)
  target_compile_definitions(libflui PRIVATE LIBFLUI_BUILD=1)
  target_include_directories(libflui PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  set_target_properties(libflui PROPERTIES OUTPUT_NAME flui CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  if(IOS)
    set_target_properties(libflui PROPERTIES FRAMEWORK TRUE MACOSX_FRAMEWORK_IDENTIFIER org.skstu.libflui MACOSX_FRAMEWORK_SHORT_VERSION_STRING "${LIBFLUI_VERSION}" MACOSX_FRAMEWORK_BUNDLE_VERSION "1")
  endif()
  install(TARGETS libflui EXPORT libfluiTargets LIBRARY DESTINATION lib RUNTIME DESTINATION bin ARCHIVE DESTINATION lib FRAMEWORK DESTINATION lib)
  install(DIRECTORY include/libflui DESTINATION include)
  install(FILES LICENSE flutter.lock.json DESTINATION share/libflui)
  install(EXPORT libfluiTargets NAMESPACE devkit:: DESTINATION lib/cmake/libflui)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/libfluiConfig.cmake" "include(\"\${CMAKE_CURRENT_LIST_DIR}/libfluiTargets.cmake\")\n")
  write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/libfluiConfigVersion.cmake" VERSION "${LIBFLUI_VERSION}" COMPATIBILITY SameMinorVersion)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/libfluiConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/libfluiConfigVersion.cmake" DESTINATION lib/cmake/libflui)
  if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_executable(flui_host_bridge_test tests/host_bridge.cpp)
    target_link_libraries(flui_host_bridge_test PRIVATE libflui)
    target_compile_features(flui_host_bridge_test PRIVATE cxx_std_17)
    # Assertions drive the bridge regression, including in Release builds.
    if(MSVC)
      target_compile_options(flui_host_bridge_test PRIVATE /UNDEBUG)
    else()
      target_compile_options(flui_host_bridge_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME libflui.host_bridge COMMAND flui_host_bridge_test)
    set_tests_properties(libflui.host_bridge PROPERTIES TIMEOUT 20)
  endif()
  return()
endif()
if(NOT APPLE OR NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  message(FATAL_ERROR "libflui v1 currently implements macOS. Other platform backends are not implemented.")
endif()
enable_language(OBJCXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(${CMAKE_CURRENT_SOURCE_DIR}/../cmake/NativeLibsBuild.cmake)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(LIBFLUI_FLUTTER_SDK "" CACHE PATH "Producer-only Flutter SDK matching flutter.lock.json")
if(NOT EXISTS "${LIBFLUI_FLUTTER_SDK}/bin/cache/flutter.version.json")
  message(FATAL_ERROR "Producer build requires LIBFLUI_FLUTTER_SDK. Binary SDK consumers do not need Flutter.")
endif()
set(LIBFLUI_ARCHS "arm64" CACHE STRING "Producer renderer architectures; first validated platform is arm64")
set(runtime "${CMAKE_CURRENT_BINARY_DIR}/lib/libflui_runtime.bundle")
set(engine "${LIBFLUI_FLUTTER_SDK}/bin/cache/artifacts/engine/darwin-x64-release/FlutterMacOS.xcframework/macos-arm64_x86_64/FlutterMacOS.framework")
file(GLOB_RECURSE renderer_inputs CONFIGURE_DEPENDS renderer/lib/*.dart renderer/test/*.dart)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/lib/renderer.complete"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/build_renderer.py"
    --sdk "${LIBFLUI_FLUTTER_SDK}" --source "${CMAKE_CURRENT_SOURCE_DIR}/renderer"
    --work "${CMAKE_CURRENT_BINARY_DIR}/renderer-work" --output "${runtime}" --arch "${LIBFLUI_ARCHS}"
  DEPENDS ${renderer_inputs} renderer/pubspec.yaml renderer/pubspec.lock flutter.lock.json tools/build_renderer.py examples/quotes/flui.xml
  VERBATIM)
add_custom_target(libflui_renderer DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/lib/renderer.complete")
add_library(libflui SHARED src/macos.mm)
add_library(devkit::libflui ALIAS libflui)
add_dependencies(libflui libflui_renderer)
target_include_directories(libflui PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_definitions(libflui PRIVATE LIBFLUI_BUILD=1)
target_compile_options(libflui PRIVATE -fobjc-arc -Wall -Wextra -Wpedantic)
target_compile_features(libflui PRIVATE cxx_std_17)
target_link_libraries(libflui PRIVATE "${engine}" "-framework Cocoa" "-framework UniformTypeIdentifiers")
set_target_properties(libflui PROPERTIES OUTPUT_NAME flui VERSION "${NATIVE_LIBS_VERSION}" SOVERSION 1
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib"
  OBJCXX_STANDARD 17 CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
  BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "@loader_path/libflui_runtime.bundle/Contents/Frameworks"
  OSX_ARCHITECTURES "${LIBFLUI_ARCHS}")
# Export only the C ABI. No Objective-C/Flutter/C++ implementation symbols.
target_link_options(libflui PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/src/exports.txt")
install(TARGETS libflui EXPORT libfluiTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/libflui DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY "${runtime}/" DESTINATION "${CMAKE_INSTALL_LIBDIR}/libflui_runtime.bundle"
  PATTERN "*.dSYM" EXCLUDE PATTERN "complete.stamp" EXCLUDE)
install(DIRECTORY examples/quotes DESTINATION ${CMAKE_INSTALL_DATADIR}/libflui/examples)
install(DIRECTORY docs DESTINATION ${CMAKE_INSTALL_DATADIR}/libflui)
install(FILES LICENSE README.md CHANGELOG.md flutter.lock.json DESTINATION ${CMAKE_INSTALL_DATADIR}/libflui)
configure_package_config_file(cmake/libfluiConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/libfluiConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/libflui)
# 0.x minor versions may change the source API; do not silently select 0.2 for 0.1.
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/libfluiConfigVersion.cmake" VERSION "${NATIVE_LIBS_VERSION}" COMPATIBILITY SameMinorVersion)
install(EXPORT libfluiTargets NAMESPACE devkit:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/libflui)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/libfluiConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/libfluiConfigVersion.cmake" cmake/BundleRuntime.cmake cmake/ConfigureApp.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/libflui)
set(LIBFLUI_RUNTIME_DIR "${runtime}")
include(cmake/BundleRuntime.cmake)
option(LIBFLUI_BUILD_SAMPLES "Build the shared-controller quote samples" ON)
option(LIBFLUI_BUILD_WXUI_SAMPLE "Also build the libwxui adapter (needs wx dependencies)" OFF)
if(LIBFLUI_BUILD_SAMPLES)
  add_subdirectory(examples/quotes)
endif()

  add_executable(flui_desktop_test tests/desktop.cpp)
  target_compile_features(flui_desktop_test PRIVATE cxx_std_20)

  add_test(NAME libflui.desktop COMMAND flui_desktop_test)
  add_executable(flui_controller_test tests/controller.cpp)
  target_compile_features(flui_controller_test PRIVATE cxx_std_17)
  add_test(NAME libflui.controller COMMAND flui_controller_test)

  libflui_bundle_runtime(flui_abi_test)
  add_test(NAME libflui.abi COMMAND flui_abi_test)
  set_tests_properties(libflui.abi PROPERTIES TIMEOUT 30)
  add_test(NAME libflui.renderer COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/build_renderer.py"
    --sdk "${LIBFLUI_FLUTTER_SDK}" --source "${CMAKE_CURRENT_SOURCE_DIR}/renderer"
    --work "${CMAKE_CURRENT_BINARY_DIR}/renderer-test" --output "${runtime}" --test)
  set_tests_properties(libflui.renderer PROPERTIES TIMEOUT 90)
